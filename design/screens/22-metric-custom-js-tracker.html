// Fires when a visitor ticks an add-on (transfer, insurance…)
splitcraft.waitForElement('.addon-list', (list) => {
  list.addEventListener('change', (e) => {
    const box = e.target.closest('input[type="checkbox"]');
    if (!box || !box.checked) return;

    splitcraft.trackEvent('add_on_selected', {
      value: Number(box.dataset.price),  // ₹ amount
      category: box.dataset.category     // "transfer"
    });
  });
});
